% Vrai ou faux ?
\begin{enumerate}
    \item Le travail d'une force dont la direction est perpendiculaire au
          déplacement de son point d'application est toujours nul.
          \dotfill \V\F
    \item La puissance du travail d'une force s'obtient en divisant son travail
          par la durée correspondante. 
          \dotfill \V\F
    \item Le travail d'une force entre deux points est toujours indépendant du
          chemin suivi.
          \dotfill \V\F
\end{enumerate}
