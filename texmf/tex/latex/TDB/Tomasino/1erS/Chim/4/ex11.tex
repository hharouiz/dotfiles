\begin{enumerate}
    \item Écrire l'équation de dissolution dans l'eau du nitrate de potassium.
    \item Calculer la conductivité d'une solution de nitrate de potassium de
          concentration molaire égale à $c=\qty{2,0e-3}{\M}$.
    \item Pour une cellule~: $S=\qty{1,0}{\square\cm}$ et $\ell=\qty{1,0}{\cm}$,
          calculer la conductance de la solution.
\end{enumerate}

\begin{donnees}
    Conductivités molaires ioniques~:
    \begin{itemize}
        \item pour le cation \ce{K^+_{(aq)}}:
              $\lambda_{\ce{K+}}=\qty{7,4e-3}{\lambda}$.
        \item pour l'anion \ce{NO$_3^-$}~:
              $\lambda_{\ce{NO$_3^-$}}=\qty{7,1e-3}{\lambda}$.
    \end{itemize}
\end{donnees}
