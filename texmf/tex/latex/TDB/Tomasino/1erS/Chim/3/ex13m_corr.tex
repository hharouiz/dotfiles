\[
    \ch{Na2CO3_{(s)} + 2 H^+_{(aq)} -> CO2_{(g)} + 2 Na^+_{(aq)} + H2O_{($\ell$)}}
\]

\begin{enumerate}
    \item
          \begin{itemize}
              \item $n_{0}(\ch{Na2CO3})=
                    \frac{m({\ch{Na2CO3}})}{M(\ch{Na2CO3})}$
                    \hfill\textbf{(0.25pt)} \\
                    avec~:
                    $M(\ch{Na2CO3})=
                    2\times M(\ch{Na})+M(\ch{C})+3\times M(\ch{O})=
                    \qty{106}{\gram\per\mole}$
          
                    A.N. $n_{0}(\ch{Na2CO3})=
                    \frac{5,3}{106}=\qty{0.050}{\mol}$.
                    \hfill\textbf{(0.25pt)}
              \item $n_0(\ch{H+})=\bigl[\ch{H+}\bigr]\cdot V=c\cdot V$
                    \hfill\textbf{(0.25pt)}
          
                    $n_0(\ch{H+})=0,8\times0,050=\qty{0,040}{\mole}$.
                    \hfill\textbf{(0.25pt)}
          \end{itemize}
    \item \mbox{} \hfill\textbf{(1pt)}
          \begin{table}[H]
              \centering
              \setlength{\arrayrulewidth}{0.8pt}
              \renewcommand{\arraystretch}{2}
              \begin{tabularx}{\textwidth}{|p{2.6cm}|
                    >{\centering\arraybackslash}p{2.2cm}|
                    >{\centering\arraybackslash}p{2.6cm}|
                    >{\centering\arraybackslash}p{2.6cm}|C|C|
                    >{\centering\arraybackslash}p{1.6cm}|}
                  \hline \multicolumn{2}{|c|}{Équation} &
                         \multicolumn{5}{c|}{\ch{Na2CO3_{(s)}} \quad + \quad
                         \ch{2 H^{+}_{(aq)}} \quad \ch{->} \quad
                         \ch{CO2_{(g)}} \quad + \quad
                         \ch{2 Na^{+}_{(aq)}} \quad + \quad
                         \ch{H2O_{($\ell$)}}} \\
                  \hline État & Avancement $x(\unit{\mol})$ &
                         quantité de \ch{Na2CO3_{(s)}} &
                         quantité d'ions \ch{H^{+}_{(aq)}} &
                         quantité d'ions \ch{Na^{+}_{(aq)}} &
                         quantité de \ch{CO2_{(g)}} &
                         quantité de \ch{H2O_{($\ell$)}} \\
                  \hline État initial &
                         $0$ & $0,050$ & $0,040$ & $0$ & $0$ & excès \\
                  \hline \parbox{3.7cm}{Au cours de la \\ transformation} &
                         $x$ & $0,050-x$ & $0,040-2x$ & $x$ & $2x$ & excès \\
                  \hline État final &
                         $x_{\max}$ & $0,050-x_{\max}$ & $0,040-2x_{\max}$ &
                         $x_{\max}$ & $2x_{\max}$ & excès \\
                  \hline
              \end{tabularx}
          \end{table}
    \item 
          \begin{itemize}
              \item Si \ch{Na2CO3_{(s)}} est le réactif limitant~: \quad
                    $0,050-x_{\max}=0 \quad\Leftrightarrow\quad
                    x_{\max}^{\mathrm{(1)}}=\qty{0.050}{\mol}$
              \item Si \ch{H^{+}_{(aq)}} est le réactif limitant~: \quad
                    $0,040-2x_{\max}=0 \quad\Leftrightarrow\quad
                    x_{\max}^{\mathrm{(2)}}=\frac{0,040}{2}=\qty{0.020}{\mol}$
          \end{itemize}
          Donc~:
          $x_{\max}=
          \min\bigl(x_{\max}^{\mathrm{(1)}}~;~x_{\max}^{\mathrm{(2)}}\bigr)=
          \qty{0.020}{\mol}$ \hfill\textbf{(0,5pt)} \\
          $\Rrightarrow$ \ch{H^{+}_{(aq)}} est le réactif limitant.
          \hfill\textbf{(0,5pt)}
    \item 
          \begin{enumerate}
              \item
                    \begin{itemize}
                        \item $n_f(\ch{Na2CO3})=0,050-x_{\max}=0,050-0,020=
                              \qty{0,030}{\mole}$
                              \hfill\textbf{(0,5pt)}
                        \item $n_f(\ch{H+})=\qty{0}{\mole}$
                              \hfill\textbf{(0,5pt)}
                        \item $n_f(\ch{CO2})=x_{\max}=\qty{0,020}{\mole}$
                              \hfill\textbf{(0,5pt)}
                        \item $n_f(\ch{Na+})=2x_{\max}=
                              2\times0,020=\qty{0,040}{\mole}$
                              \hfill\textbf{(0,5pt)}
                    \end{itemize}
              \item $V(\ch{CO2})=n_f(\ch{CO2})\times V_m$
                    \hfill\textbf{(0,5pt)} \\
                    A.N. \ $V(\ch{CO2})=0,020\times24=\qty{0,48}{\liter}$
                    \hfill\textbf{(0,5pt)}
              \item $[\ch{Na+}]=\frac{n_f(\ch{Na+})}{V}$
                    \hfill\textbf{(0,5pt)} \\[3mm]
                    $[\ch{Na+}]=\frac{0,040}{0,050}=\qty{0,80}{\mole\per\liter}$
                    \hfill\textbf{(0,5pt)}
          \end{enumerate}
\end{enumerate}
