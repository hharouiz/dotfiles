% Vrai ou faux~?
Soit l'équation d'une réaction et les quantité de matières initiales~:

\[
    \begin{array}{lccccccc}
        \text{É.I~:} & \ch{Al2 S3_{(s)} &+& 3 H2O_{($\ell$)} &->& 3 H2S_{(g)} &+& Al2O3} \\
                     & \qty{1}{\mol} && \qty{2}{\mol} && 0 && 0 \\
    \end{array}
\]
\begin{enumerate}
    \item Le réactif limitant est l'eau \dotfill \V\F
    \item À la fin de la réaction, il s'est formé dans le mélange
          $\qty{2}{\mol}$ \ch{H2S} \dotfill \V\F
    \item À la fin de la réaction, il s'est formé dans le mélange
          $\qty{1}{\mol}$ \ch{Al2O3} \dotfill \V\F
    \item À la fin de la réaction, il reste dans le mélange $\qty{0.33}{\mol}$
          de \ch{Al2S3} \dotfill \V\F
\end{enumerate}
