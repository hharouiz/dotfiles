% Équations de réactions
Soient les couples oxydant/réducteur suivants~:

\ce{Zn$^{2+}$_{(aq)} / Zn_{(s)}}~;
\ce{H$^+$_{(aq)} / H2_{(g)}}~;
\ce{MnO$_4^{-}$_{(aq)} / Mn^{2+}_{(aq)}}~;
\ce{O2_{(g)} / H2O2_{(aq)}}~;
\ce{I2_{(aq)} / I$^-$_{(aq)}}~;
\ce{S2O$_8^{2-}$_{(aq)} / SO$_4^{2-}$_{(aq)}}.

\begin{enumerate}
    \item Écrire l'équation d'oxydoréduction entre les espèces~:
          \ce{Zn_{(s)}} et \ce{H$^+$_{(aq)}}.
          \textbf{(1pt)}
    \item Même question pour les espèces \ce{MnO$_4^{-}$_{(aq)}} et
          \ce{H2O2_{(aq)}}.
          \textbf{(1pt)}
    \item Même question pour les espèces \ce{I$^-$_{(aq)}} et
          \ce{S2O$_8^{2-}$_{(aq)}}.
          \textbf{(1pt)}
\end{enumerate}
