% Molécule d'ammoniac
\begin{enumerate}
    \item Les liaisons \chemfig{N-H} sont polarisées car \ch{N} est plus
          électronégatif que \ch{H}.~\textbf{(0,5pt)}
    \item La molécule d'ammoniac est polaire, car le barycentre des charges
          positives ne coïncide pas avec le barycentre des charges
          négatives.~\textbf{(0,5pt)}
    \item \mbox{}\par
          \begin{center}
              \chemfig{\charge{90=\|,90[anchor=180+\chargeangle,
                  inner ysep=3pt]=\small$3\delta^{-}$}{N}(-[:225]\charge{-90[anchor=180+\chargeangle,
                  inner ysep=3pt]=\small$\delta^{+}$}{H})(<[:-90]\charge{-90[anchor=180+\chargeangle,
                  inner ysep=3pt]=\small$\delta^{+}$}{H})(<:[:-45]\charge{-90[anchor=180+\chargeangle,
                  inner ysep=3pt]=\small$\delta^{+}$}{H})}
          \end{center}
          \textbf{(0,5pt)}
\end{enumerate}
