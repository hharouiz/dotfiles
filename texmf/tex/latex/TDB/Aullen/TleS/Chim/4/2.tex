$11,62$.

    \begin{itemize}
        \item Produit ionique de l'eau~: $\mathrm{K_{e}}=\num{1,0e-14}$.
    \end{itemize}
\end{donnees}

\begin{enumerate}
    \item Déterminer la concentration en ions oxonium de $\mathrm{S_{1}}$.
    \item Montrer que la concentration en ions hydroxyde dans la solution
          $\mathrm{S_{1}}$ est $\bigl[\ce{HO-}\bigr]=\qty{4,2e-3}{\M}$.
    \item Quels sont les ions prédominants~? Est-ce en accord avec le pH de la
        solution~?
\end{enumerate}
