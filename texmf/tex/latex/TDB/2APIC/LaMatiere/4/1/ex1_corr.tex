La combustion du \textbf{butane} dans le dioxygène se fait avec une flamme
\textbf{bleue} et donne deux produits qu'on a prouvé par~:
\begin{itemize}
    \item \textbf{Le produit~(a)~:} Des petites gouttes de l'eau sur les parois
          du tube à essai.
    \item \textbf{Le produit~(b)~:} trouble l'eau chaux.
\end{itemize}
\begin{enumerate}
    \item En justifiant ta réponse, cette combustion est-t-elle
          \textbf{complète} ou bien \textbf{incomplète}~?
          \textbf{(0.5pt)} \par
          \fillblank*{1cm}{Complète, car la flamme est bleue}
    \item Comment est la quantité de dioxygène (\textbf{grande} ou
          bien \textbf{petite})~?
          \textbf{(0.5pt)} \par
          \fillblank*{1cm}{Grande}
    \item Quels sont les \textbf{réactifs} de cette réaction de combustion~?
          \textbf{(0,5pt)} \par
          \fillblank*{1cm}{Butane et dioxygène}
    \item Détermine le combustible et le comburant pendant cette réaction
          chimique~:
          \textbf{(0.5pt)} \par
          \begin{itemize}
              \item \textbf{Le combustible~:} \fillblank*{1cm}{butane}
              \item \textbf{Le comburant~:} \fillblank*{1cm}{dioxygène}
          \end{itemize}
    \item Identifie les produits de cette réaction de combustion~?
          \textbf{(1pt)} \par
          \begin{itemize}
              \item \textbf{Le produit~(a)~:} \fillblank*{1cm}{eau}
              \item \textbf{Le produit~(b)~:} \fillblank*{1cm}{dioxygène de
                    carbone}
          \end{itemize}
    \item Écrire la réaction bilan de cette réaction~?
          \textbf{(1pt)} \par
          \fillblank*{1cm}{Butane + dioxygène $\rightarrow$ eau +
          dioxyde de carbone}
    \item Si la quantité de l'air est \textbf{insuffisante} (petite) pour cette
          réaction de combustion
          \begin{enumerate}
              \item Que sera-t-il le type de la réaction (\textbf{complète} ou
                    \textbf{incomplète})~?
                    \textbf{(0,5pt)} \par
                    \fillblank*{1cm}{Incomplète}
              \item Que sera-t-elle la couleur de la flamme~?
                    \textbf{(0,5pt)} \par
                    \fillblank*{1cm}{Jaune}
              \item Écrire la réaction bilan de cette réaction.
                    \textbf{(1pt)} \par
                    \fillblank*{1cm}{Butane + dioxygène $\rightarrow$ eau +
                    Dioxyde de carbone + monoxyde de carbone + carbone}
          \end{enumerate}
\end{enumerate}
