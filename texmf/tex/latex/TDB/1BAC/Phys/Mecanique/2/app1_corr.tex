\begin{enumerate}
    \item
         \begin{itemize}
             \item $\vec{P}$~: le poids
             \item $\vec{T}$~: la tension du fil
             \item $\vec{R}$~: la réaction du sol
         \end{itemize}
         \begin{figure}[H]
            \centering
            \begin{tikzpicture}
                \def\a{20}
                \def\b{10}
                \def\F{1.2}
                \def\AC{6.3}
                \pgfmathparse{\AC*tan(\b)}\let\BC=\pgfmathresult
                \coordinate (A) at (0,0);
                \path[fill=black!10,draw,thick]
                    (A) -- ++(\AC,0) coordinate (C) -- ++(0,\BC)
                    coordinate (B) -- cycle;
                \node[solide={1cm/7mm},rotate=\b,anchor=south,
                    outer ysep=1.2pt] (S) at ($(A)!0.5!(B)$) {};
                \node[circle,fill,inner sep=0.8pt] (F) at (S.east) {};
                \draw[dashed] (F.center) -- ++(\b:2.8);
                \draw[thick] (F.center) -- ++(\b+\a:3);
                \draw[->,shorten >=0.4pt]
                    ([xshift=2cm]A) arc (0:\b:2cm)
                    node[pos=0.5,right,font=\footnotesize] {$\beta$};
                \draw[->,shorten >=0.4pt]
                    ([shift=(\b:1)]F.center) arc (\b:\b+\a:1)
                    node[pos=0.7,right,font=\footnotesize] {$\alpha$};
                % corr
                \draw[->,very thick,red] (S.south) -- ++(90+\b:2)
                    node[right,pos=0.9] {$\vec{R}$};
                \draw[->,very thick,blue] (S.center) -- ++(270:2)
                    node[right,pos=0.8] {$\vec{P}$};
                \draw[->,very thick,Green] (S.east) -- ++(\a+\b:2)
                    node[above,pos=0.8] {$\vec{T}$};
            \end{tikzpicture}
        \end{figure}
    \item
        \begin{itemize}
            \item Travail de la tension du fil sur le trajet~:
                  \begin{align*}
                      W_{AB}(\vec{T})&=\vec{T}\cdot\vv{AB}=
                      T\times AB\times\cos\alpha \\
                      &=9,2\times20\times\cos(\ang{20})=
                      \qty{1.7e2}{\J}
                  \end{align*}
                  
                  Le travail de la tension du fil est positif, la tension du fil
                  est une force motrice.
            \item Travail du poids sur le trajet~:
                  \begin{align*}
                      W_{AB}(\vec{P})&=\vec{P}\cdot\vv{AB}=
                      m\times g\times AB\times\cos(\ang{90}+\beta) \\
                      &=2,0\times9,8\times20\times\cos(\ang{100})=
                      \qty{-68}{\J}
                  \end{align*}
                  
                  Le travail du poids est négatif, le poids est une force
                  résistante (il s'oppose au mouvement).
            \item Travail de la réaction du sol~:
                  \[
                      W_{AB}(\vec{R})=\vec{R}\cdot\vv{AB}=
                      R\times AB\times\cos(\ang{90})=\qty{0}{\J}
                  \]
                  
                  Le travail de la réaction du sol est nul car on a négligé les
                  frottements~: la réaction du sol est une force qui ne
                  travaille pas.
        \end{itemize}
\end{enumerate}
