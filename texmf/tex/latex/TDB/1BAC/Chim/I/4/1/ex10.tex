    \item On considère les espèces chimiques suivantes~:
          \begin{itemize}
              \item L'acide éthanoïque $\ce{CH3COOH}$.
              \item L'ion ammoniac $\ce{NH3}$ (base).
          \end{itemize}
          \begin{enumerate}
              \item Écrire les demi-équations électroniques des couples
                    acide/base ci-dessus.~\textbf{(1pt)}
              \item Écrire l'équation de la réaction entre l'acide éthanoïque et
                    l'ion ammoniac.~\textbf{(0,5pt)}
          \end{enumerate}
    \item On donne l'équation suivante~:
          $\ce{HF + H2O -> F- + H3O+}$.

          Identifier les deux couples acide/base mis en jeu et écrire les
          demi-équations correspondantes.~\textbf{(1pt)}
\end{enumerate}
