    \item $C=\frac{n}{V}=\frac{m}{MV}=\frac{0,271}{162,5\times0,250}=

          \hfill\textbf{(1pt)}

          \lambda(\ce{Cl^-})\cdot\bigl[\ce{Cl^-}\bigr]$

          $[\ce{Fe^{3+}}]=\qty{6,67}{\mol\per\cubic\m}, \quad
           [\ce{Cl^-}]=\qty{20,0}{\mol\per\cubic\m}$

          $\sigma=\num{2,04e-2}\times\qty{6,67}+\num{7,63e-3}\times\num{20,0}=
          \qty{0,289}{\S\per\m}$
          \hfill\textbf{(1pt)}
