La densité du fer est $d=7,8$.
\begin{enumerate}
    \item Calculer la masse d'un cube de fer d'arête $a=\qty{20}{\cm}$.
    \item Calculer la quantité de matière d'atomes de fer contenus dans ce cube.
\end{enumerate}

\begin{donnees}
    $\rho_{\text{eau}}=\qty{1000}{\kg\per\cubic\meter}$, \
    $M(\ce{Fe})=\qty{55,8}{\gram\per\mol}$.
\end{donnees}
