\begin{enumerate}
    \item $P_1V=n_1RT \quad\Rightarrow n_1=\frac{P_1V}{RT}$
          \hfill\textbf{(0,5pt)}
          \\
          $\Rrightarrow n_{1}=\frac{\num{1013e2}\times2}{8,314\times298}
          \approx\qty{81.77}{\mol}$
          \hfill\textbf{(0,5pt)}
    \item $P_1^{\prime}V^{\prime}=n_1RT \quad\Rightarrow
          P_1^{\prime}=\frac{n_1 R T}{V^{\prime}}$
          \hfill\textbf{(0,5pt)}
          \\[2mm]
          $\Rrightarrow P_{1}^{\prime}=
           \frac{81,77\times8,314\times298}{\num{20e-3}}=
           \qty{10130}{\kPa}$
          \hfill\textbf{(0,5pt)}
    \item 
          \begin{itemize}
              \item $n_{\text{tot}}=\frac{P_{2}V}{RT}$ \\
                    $\Rrightarrow n_{\text{tot}}=
                    \frac{\num{1040e2}\times2}{8,314\times298}
                    \approx\qty{83.95}{\mol}$
                    \hfill\textbf{(0,25pt)}
              \item $n_{\ce{CO2}}=n_{\text{tot}}-n_{1}$ \\
                    $\Rrightarrow n_{\ce{CO2}}=83,95-81,77=\qty{2.18}{\mol}$
                    \hfill\textbf{(0,25pt)}
              \item $m_{\text{tot}}=
                    n_{1}\times M_{\ce{O2}}+n_{\ce{CO2}}\times M_{\ce{CO2}}$ \\
                    $\Rrightarrow m_{\text{tot}}=81,77\times32+2,18\times44
                    \approx\qty{2713}{\g}$
                    \hfill\textbf{(0,5pt)}
          \end{itemize}
\end{enumerate}
