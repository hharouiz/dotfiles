\begin{enumerate}
    \item Un acide de Brönsted est une espèce chimique capable de céder un
          proton \ch{H+}.
          \hfill\textbf{(1pt)}
    \item $\ch{C3H7COOH_{(aq)} + H2O <=> C3H7COO^{-}_{(aq)} + H3O+}$
          \hfill\textbf{(1pt)}
    \item $n_{0}=C\times V$
          \hfill\textbf{(0.5pt)}

          $n_{0}=\num{2.0e-3}\times\num{1.0}
          \quad\Rightarrow\quad
          \mathbox{n_{0}=\qty{2.0e-3}{\mol}}$
          \hfill\textbf{(0.5pt)}
    \item Tableau d'avancement~: \hfill\textbf{(1pt)}

          \begin{minipage}{\linewidth}
              \begin{table}[H]
              \centering
              \renewcommand{\arraystretch}{2}
              \begin{tabularx}{\textwidth}{|p{3.0cm}|
                    >{\centering\arraybackslash}p{2.3cm}|
                    >{\centering\arraybackslash}p{3.1cm}|
                    >{\centering\arraybackslash}p{2.0cm}|C|C|}
                  \hline \multicolumn{2}{|c|}{Équation} &
                  \multicolumn{4}{c|}{\ch{C3H7COOH_{(aq)}} \quad $+$ \quad
                         \ch{H2O} \ \ch{->} \
                         \ch{C3H7COO^{-}_{(aq)}} \quad $+$ \quad
                         \ch{H3O+}} \\
                  \hline État & Avancement $x~(\unit{\mol})$ &
                         $n(\ch{C3H7COO^{-}})$ & $n(\ch{H2O})$ &
                         $n(\ch{C3H7COO^{-}})$ & $n(\ch{H3O+})$ \\
                  \hline État initial &
                         $0$ & $n_{0}$ & excès & $0$ & $0$ \\
                  \hline \parbox{3.7cm}{Au cours de la \\ transformation} &
                         $x$ & $n_{0}-x$ & excès & $x$ & $x$ \\
                  \hline État final & 
                         $x_{f}$ & $n_{0}-x_{f}$ & excès &
                         $x_{f}$ & $x_{f}$ \\
                  \hline
              \end{tabularx}
          \end{table}
          \end{minipage}
    \item L'eau est en excès. \ch{C3H7COOH_{(aq)}} est le réactif limitant.

          $n_{0}-x_{\max}=0 \quad\Rightarrow\quad
          x_{\max}=n_{0}=\qty{2,0e-3}{\mol}$
          \hfill\textbf{(1pt)}
    \item D'après le tableau d'avancement~: \
          $x_{f}=n(\ch{H3O+})=\left[H_{3} O^{+}\right]\cdot V$
          \hfill\textbf{(0.5pt)}
        
          $x_{f}=10^{-3,76} \times 1,0 \quad\Rightarrow\quad
          \mathbox{x_{f}=\qty{1,74e-4}{\mol}}$
          \hfill\textbf{(0.5pt)}
    \item $\tau=\frac{x_{f}}{x_{\max}}=\frac{\num{1.74e-4}}{\num{2.0e-3}}=0,087
          \quad\Rightarrow\quad
          \mathbox{\tau=8,7\%}$
          \hfill\textbf{(0.75pt)}

          $\tau<1\ \Rightarrow$ La réaction est limitée.
          \hfill\textbf{(0.25pt)}
\end{enumerate}
