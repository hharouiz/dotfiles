\begin{enumerate}
    \item 
          \begin{enumerate}
              \item L'onde mécanique progressive périodique est dite sinusoïdale
                    si la perturbation est une fonction sinusoïdale du temps. 
                    \textbf{(1pt)}
              \item L'onde est transversale car la direction de la perturbation
                    est perpendiculaire au sens de déplacement de l'onde.
                    \textbf{(1pt)}
          \end{enumerate}
    \item 
          \begin{enumerate}
              \item $\lambda=\frac{\qty{4}{\cm}}{2}=\qty{2}{\cm}$.
                    \textbf{(1pt)}
              \item $v=\lambda\times f=\num{2e-2}\times20=\qty{0.4}{\v}$.
                    \textbf{(1pt)}
              \item On a~:
                    $v=\frac{d}{\Delta t}=\frac{d}{t_{1}-t_{0}}=
                    \frac{d}{t_{1}}$, donc~:
                    $t_1=\frac{d}{v}=\frac{\num{4e-2}}{\num{0.4}}=\qty{0.1}{s}$.
                    \textbf{(1pt)}
          \end{enumerate}

      \item $y_{M}(t)=y_{S}(t-t_{1})$ $\Rightarrow y_{M}(t)=y_{S}(t-0,1)$.
          \textbf{(1pt)}
  
    \item
          \begin{itemize}
              \item Pour $f^{\prime}=\qty{10}{\Hz}$, $\lambda^{\prime}=\qty{3}{\cm}$,
                    $v^{\prime}=\lambda^{\prime}f^{\prime}=\qty{0.3}{\v}$.  
              \item Pour $f=\qty{20}{Hz}$, $\lambda=\qty{2}{\cm}$,
                    $v=\qty{0.4}{\v}$.  
          \end{itemize}
          $v^{\prime}\neq v$ $\Rightarrow$ la célérité dépend de la fréquence
          $\Rightarrow$ le milieu de propagation est \textbf{dispersif}.
          \textbf{(1pt)}
    \item 
          \begin{enumerate}
              \item $f=\qty{20}{\Hz}$, $v=\qty{0.4}{\v}$ $\Rightarrow$
                    longueur d'onde $\lambda=\qty{2}{\cm}$.

                    $a<\lambda$ $\Rightarrow$ il y a diffraction.
                    \textbf{(1pt)}
              \item Après la traversée de l'ouverture, l'onde devient
                    \textbf{circulaire} mais garde sa \textbf{même longueur
                    d'onde} $\lambda=\qty{2}{\cm}$.
                    \textbf{(1pt)}
          \end{enumerate}
\end{enumerate}
