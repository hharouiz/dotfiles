    \item On a~: $\tan\theta=\frac{L/2}{D}=\frac{L}{2D}$.
          Pour $\theta$ petite, $\tan\theta\approx\theta$, donc~:
          $\theta=\frac{L}{2 D}$.
          \hfill\textbf{(0,75pt)}
    \item On a~: $\theta=\frac{\lambda}{a}=\frac{L}{2 D}$, donc~:
          $D=\frac{L \cdot a}{2\lambda}$.
          \hfill\textbf{(0,5pt)}

          A.N. \ $D=\frac{\num{3.8e-2}\times\num{5e-5}}{2\times\num{633e-9}}=
          \qty{1.5}{\meter}$.
          \hfill\textbf{(0,25pt)}
    \item On a~: $c=\lambda f$, donc~: $f=\frac{c}{\lambda}$.
          \hfill\textbf{(0,5pt)}

          A.N. \ $f=\frac{\num{3e8}}{\num{633e-9}}=\qty{4.74e14}{\hertz}$.
          \hfill\textbf{(0,5pt)}
    \item On a~: $D=\frac{L_{0} \cdot a_{0}}{2\lambda}$, donc~:
          $a_{0}=\frac{2\lambda\cdot D}{L_{0}}$.
          \hfill\textbf{(0,5pt)}

          A.N. \ $a_{0}=\frac{2\times\num{633e-9}\times 1,5}{\num{1.5e-2}}=
          \qty{1.3e-4}{\meter}$.
